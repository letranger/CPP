#+begin_src cpp -r -n :results output :exports both
#include <iostream>
using namespace std;
int main(){
    int x;
    int y;
    cin>>x>>y;
    int n[10000];
    int p;
    int q;
    // 註解：for 迴圈
    for(int i=0;i<x;i++){
        for(int j=0;j<y;j++){
            p = i*y + j;
            cin>>n[p];
        }
    }

    for(int a=0;a<y;a++){
        for(int b=0;b<x;b++){
            q = b*y + a;
            cout<<n[q];
            if(b<(x-1)){
                cout<<" ";
            }
        }
        cout<<"\n";
    }
    return 0;
}
#+end_src
